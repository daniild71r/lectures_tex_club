\begin{definition}
    Случайный процесс $\{\xi_n\}_{n \in \Z_+}$ называется \textit{предсказуемым}
    относительно фильтрации $\{\F_n\}_{n \in \Z_+}$, если $\xi_n$ 
    оказывается $\F_{n - 1}$-измеримым для всех $n$.
\end{definition}

\begin{note}
    Это свойство сильнее согласованности.
\end{note}

\begin{lemma}
    Пусть $\{\xi_n\}_{n \in \Z_+}$~--- мартингал относительно $\{\F_n\}$ и $\xi_0 = 0$. Если $\{\xi_n\}$ предсказуем, то $\xi_n \equiv 0$.
\end{lemma}

\begin{proof}
    Проверим, что $\xi_1 = 0$.

    Из мартингальности $\E(\xi_1 | \F_0) = \xi_0 = 0$. Так как $\xi_1$ измерима относительно $\F_0$ (из предсказуемости), получаем $\E(\xi_1 | \F_0) = \xi_1$.

    Аналогичные рассуждения проводим для всех $\xi_n$.
\end{proof}

\begin{theorem}[разложение Дуба-Мейера]
    Пусть процесс $\{\xi_n\}_{n \in \Z_+}$ согласован с фильтрацией $\{\F_n\}_{n \in \Z_+}$, причем все $\xi_n$ интегрируемы. Тогда существует единственное с точностью до эквивалентности разложение

    \[
    \xi_n = \mu_n + \pi_n,
    \]

    где $\{\mu_n\}$~--- мартингал относительно этой фильтрации, а процесс $\{\pi_n\}$ предсказуем, $\pi_0 = 0$.
\end{theorem}

\begin{proof}
     Положим $\pi_0 = 0$, а при $n \geq 1$ определим по правилу:

     \[
     \pi_n = \sum_{k = 1}^n \E(\xi_k - \xi_{k - 1} | \F_{k - 1})
     \]

    По построению процесс $\{\pi_n\}$ предсказуем. Осталось доказать, что
    $\mu_n = \xi_n - \pi_n$ является мартингалом. $\xi_n$ и $\pi_n$ интегрируемы, поэтому $\mu_n$ интегрируема. Согласованность $\{\mu_n\}$ с $\{\F_n\}$ тоже есть. Осталось проверить, что $\E(\mu_n | \F_{n - 1}) = \mu_{n - 1}$.

    \[
    \E(\mu_n | \F_{n - 1}) = \E(\xi_n | \F_{n - 1}) - \pi_n =
    \E(\xi_{n - 1} | \F_{n - 1}) - \sum_{k = 1}^{n - 1} \E(\xi_k - \xi_{k - 1} | \F_{k - 1}) = \xi_{n - 1} - \pi_{n - 1} = \mu_{n - 1}
    \]

    Покажем единственность разложения. Пусть $\xi_n = \mu_n' + \pi_n'$~--- ещё одно разложение с указанными свойствами. Тогда 
    $\mu_n + \pi_n = \mu_n' + \pi_n' \lra \mu_n - \mu_n' = \pi_n' - \pi_n$. Обозначим этот процесс за $y_n$.
    $\mu_n - \mu_n'$~--- мартингал, $\pi_n - \pi_n'$~--- предсказуемый, $\pi_0' = 0$. Тогда к $y_n$ можно применить лемму, отсюда получаем, что $y_n \equiv 0$,
    а значит $\mu_n \equiv \mu_n'$ и $\pi_n \equiv \pi_n'$.
\end{proof}
